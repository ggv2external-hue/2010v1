local function safeTween(obj, info, props)

local function resolveColor(c)
    if type(c) == "function" then
        local ok, res = pcall(c)
        if ok then return res end
        return nil
    end
    return c
end

-- ============================ PLANE CACHE ============================
local cachedPlaneParts = {}
local cachedPlaneModels = {}
local lastPartCacheUpdate = 0

local function modelHasPlaneParts(m)
    if not m or not m:IsA("Model") then return false end
    if m:FindFirstChild("Fuselage") then return true end
    if m.PrimaryPart then return true end
    return false
end

local function playerPlaneModel(plr)
    local direct = Workspace:FindFirstChild(plr.Name)
    if modelHasPlaneParts(direct) then return direct end
    for _, obj in ipairs(Workspace:GetChildren()) do
        if obj:IsA("Model") and obj.Name == plr.Name and modelHasPlaneParts(obj) then return obj end
    end
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("Model") and modelHasPlaneParts(obj) then
            if obj.Name == plr.Name or obj.Name:find(plr.Name, 1, true) then return obj end
        end
    end
    return nil
end

local function FindAllPlaneParts()
    local now = os.clock()
    if now - lastPartCacheUpdate > 0.5 then
        lastPartCacheUpdate = now
        table.clear(cachedPlaneParts)
        table.clear(cachedPlaneModels)
        local seen, seenModels = {}, {}

        local named = {}
        for _, obj in ipairs(Workspace:GetDescendants()) do
            if obj:IsA("BasePart") then
                local n = obj.Name
                if n == "Fuselage" or n == "LeftWing" or n == "RightWing" then
                    local m = obj:FindFirstAncestorWhichIsA("Model")
                    if m then
                        named[m] = named[m] or {model = m, fuselage = nil, wings = {}, isBot = false}
                        if n == "Fuselage" then named[m].fuselage = obj end
                        if n == "LeftWing" or n == "RightWing" then table.insert(named[m].wings, obj) end
                    end
                    if not seen[obj] then seen[obj] = true; table.insert(cachedPlaneParts, obj) end
                end
            end
        end
        for m, data in pairs(named) do
            if not seenModels[m] then
                seenModels[m] = true
                if not data.fuselage and m.PrimaryPart then data.fuselage = m.PrimaryPart end
                table.insert(cachedPlaneModels, data)
            end
        end
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LP then
                local m = playerPlaneModel(plr)
                if m and not seenModels[m] then
                    seenModels[m] = true
                    local data = {model = m, fuselage = m:FindFirstChild("Fuselage") or m.PrimaryPart, wings = {}, isBot = false}
                    for _, n in ipairs({"LeftWing", "RightWing"}) do
                        local p = m:FindFirstChild(n)
                        if p and p:IsA("BasePart") then
                            table.insert(data.wings, p)
                            if not seen[p] then seen[p] = true; table.insert(cachedPlaneParts, p) end
                        end
                    end
                    if data.fuselage and not seen[data.fuselage] then
                        seen[data.fuselage] = true; table.insert(cachedPlaneParts, data.fuselage)
                    end
                    table.insert(cachedPlaneModels, data)
                end
            end
        end
        pcall(function()
            local PS = LP:FindFirstChild("PlayerScripts")
            local CC = PS and PS:FindFirstChild("ClientCore")
            local FW = CC and CC:FindFirstChild("Framework")
            local MD = FW and FW:FindFirstChild("Modules")
            local CM = MD and MD:FindFirstChild("Camera")
            if CM then
                local cam = require(CM)
                if cam then
                    for _, fn in ipairs({"plane_viewmodels", "plane_part_viewmodels", "bot_plane_viewmodels"}) do
                        local f = cam[fn]
                        if f then
                            for _, child in ipairs(f:GetChildren()) do
                                if child:IsA("Model") and not seenModels[child] then
                                    seenModels[child] = true
                                    local data = {model = child, fuselage = child:FindFirstChild("Fuselage") or child.PrimaryPart, wings = {}, isBot = (fn == "bot_plane_viewmodels")}
                                    for _, n in ipairs({"LeftWing", "RightWing"}) do
                                        local p = child:FindFirstChild(n)
                                        if p and p:IsA("BasePart") then table.insert(data.wings, p) end
                                    end
                                    if data.fuselage then table.insert(cachedPlaneModels, data) end
                                end
                            end
                        end
                    end
                end
            end
        end)
    end
    return cachedPlaneParts, cachedPlaneModels
end

-- ============================ DRAWING POOL ============================
local DrawingPool = {
    Lines = {}, Squares = {}, Circles = {}, Texts = {},
    Indexes = {Line = 1, Square = 1, Circle = 1, Text = 1}
}

local function ResetDrawingPool()
    DrawingPool.Indexes.Line = 1
    DrawingPool.Indexes.Square = 1
    DrawingPool.Indexes.Circle = 1
    DrawingPool.Indexes.Text = 1
    for _, obj in pairs(DrawingPool.Lines) do obj.Visible = false end
    for _, obj in pairs(DrawingPool.Squares) do obj.Visible = false end
    for _, obj in pairs(DrawingPool.Circles) do obj.Visible = false end
    for _, obj in pairs(DrawingPool.Texts) do obj.Visible = false end
end

local function GetPooledDrawing(kind)
    local idx = DrawingPool.Indexes[kind]
    DrawingPool.Indexes[kind] = idx + 1
    local pool = DrawingPool[kind .. "s"]
    local obj = pool[idx]
    if not obj then
        obj = Drawing.new(kind)
        pool[idx] = obj
    end
    obj.Visible = true
    return obj
end

-- ============================ CORNER + STROKE ============================
local cornerRegistry = {}

local function corner(obj, scale)
    scale = scale or 1
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, math.max(0, math.floor(Config.UICornerRadius * scale)))
    c.Parent = obj
    if scale <= 5 then table.insert(cornerRegistry, {corner=c, scale=scale}) end
    return c
end

local function cornerAbs(obj, r)
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, r or 6)
    c.Parent = obj
    return c
end

local function stroke(obj, color, thickness)
    local s = Instance.new("UIStroke")
    s.Color = color or Theme.Border
    s.Thickness = thickness or 1
    s.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    s.Parent = obj
    return s
end

local function RepaintCorners()
    local base = Config.UICornerRadius
    for _, e in ipairs(cornerRegistry) do
        if e.corner and e.corner.Parent then
            e.corner.CornerRadius = UDim.new(0, math.max(0, math.floor(base * e.scale)))
        end
    end
    if _G.__WOG_WM_CORNER then
        _G.__WOG_WM_CORNER.CornerRadius = UDim.new(0, math.max(0, math.floor(base * (_G.__WOG_WM_SCALE or 1))))
    end
    if _G.__WOG_WMSHADOW_CORNER then
        _G.__WOG_WMSHADOW_CORNER.CornerRadius = UDim.new(0, math.max(0, math.floor(base * (_G.__WOG_WM_SHADOW_SCALE or 1))))
    end
end

local RepaintRegistry = {
    sliders = {}, toggles = {}, buttons = {}, sections = {}, accentLabels = {}, labels = {},
}

local function regSlider(fill, handleStroke) table.insert(RepaintRegistry.sliders, {fill=fill, handleStroke=handleStroke}) end
local function regToggle(pill, knob, accentBar, getState) table.insert(RepaintRegistry.toggles, {pill=pill, knob=knob, accentBar=accentBar, getState=getState}) end
local function regButton(btn, colorFn) table.insert(RepaintRegistry.buttons, {btn=btn, colorFn=colorFn}) end
local function regSection(bullet, label) table.insert(RepaintRegistry.sections, {bullet=bullet, label=label}) end
local function regAccentLabel(obj) table.insert(RepaintRegistry.accentLabels, obj) end
local function regLabel(obj, getKey) table.insert(RepaintRegistry.labels, {obj=obj, key=getKey}) end

local function RepaintTheme()
    for _, e in ipairs(RepaintRegistry.sliders) do
        if e.fill and e.fill.Parent then
            e.fill.BackgroundColor3 = Theme.Accent
            local grad = e.fill:FindFirstChildOfClass("UIGradient")
            if grad then grad.Color = ColorSequence.new(Theme.AccentDim, Theme.Accent) end
        end
        if e.handleStroke and e.handleStroke.Parent then e.handleStroke.Color = Theme.Accent end
    end
    for _, e in ipairs(RepaintRegistry.toggles) do
        local on = e.getState and e.getState() or false
        if e.pill and e.pill.Parent then e.pill.BackgroundColor3 = on and Theme.Accent or Theme.Surface3 end
        if e.accentBar and e.accentBar.Parent then e.accentBar.BackgroundColor3 = Theme.Accent end
    end
    for _, e in ipairs(RepaintRegistry.buttons) do
        if e.btn and e.btn.Parent then
            local base = e.colorFn and e.colorFn() or nil
            if base then e.btn.BackgroundColor3 = base end
        end
    end
    for _, e in ipairs(RepaintRegistry.sections) do
        if e.bullet and e.bullet.Parent then e.bullet.BackgroundColor3 = Theme.Accent end
    end
    for _, lbl in ipairs(RepaintRegistry.accentLabels) do
        if lbl and lbl.Parent then lbl.TextColor3 = Theme.Accent end
    end
    for _, e in ipairs(RepaintRegistry.labels) do
        if e.obj and e.obj.Parent and e.key then
            e.obj.Text = T(e.key)
        end
    end
end

local function RepaintUI()
    pcall(function()
        -- top bar accent, brand dot, outer glow
        if _G.__WOG_TOPBAR_ACCENT then _G.__WOG_TOPBAR_ACCENT.BackgroundColor3 = Theme.Accent end
        if _G.__WOG_BRAND_DOT then _G.__WOG_BRAND_DOT.BackgroundColor3 = Theme.Accent end
        if _G.__WOG_BRAND_DOT_STROKE then _G.__WOG_BRAND_DOT_STROKE.Color = Theme.AccentHi end
        if _G.__WOG_OUTER_GLOW then _G.__WOG_OUTER_GLOW.BackgroundColor3 = Theme.Accent end
        if _G.__WOG_MINI_PILL then
            _G.__WOG_MINI_PILL.TextColor3 = Theme.Text
            local s = _G.__WOG_MINI_PILL:FindFirstChildOfClass("UIStroke")
            if s then s.Color = Theme.Accent end
        end
    end)
    RepaintTheme()
    RepaintCorners()
end

local activeNotifCount = 0

local function dismissNotif(entry)
    if not entry or entry.dismissed then return end
    entry.dismissed = true
    local card = entry.card
    local isRight = Config.NotifPosition:find("right") ~= nil
    local offX = isRight and (Config.NotifWidth + 60) or -(Config.NotifWidth + 60)
    safeTween(card, TweenInfo.new(0.28, Enum.EasingStyle.Quart, Enum.EasingDirection.In), {
        Position = UDim2.new(0, offX, 0, 0), BackgroundTransparency = 1,
    })
    for _, d in ipairs(card:GetDescendants()) do
        if d:IsA("TextLabel") then safeTween(d, TweenInfo.new(0.22), { TextTransparency = 1 })
        elseif d:IsA("Frame") or d:IsA("TextButton") then safeTween(d, TweenInfo.new(0.22), { BackgroundTransparency = 1 })
        elseif d:IsA("UIStroke") then safeTween(d, TweenInfo.new(0.22), { Transparency = 1 }) end
    end
    activeNotifCount = math.max(0, activeNotifCount - 1)
    task.delay(0.32, function() if entry.wrapper and entry.wrapper.Parent then entry.wrapper:Destroy() end end)
    for i = #State.NotifActive, 1, -1 do
        if State.NotifActive[i] == entry then table.remove(State.NotifActive, i) end
    end
    if #queuedNotifs > 0 then
        local n = table.remove(queuedNotifs, 1)
        task.defer(function() SendNotification(n.title, n.message, n.duration, n.kind) end)
    end
end

function SendNotification(title, message, duration, kind)

    duration = duration or Config.NotifDuration
    kind = kind or "info"
    title = tostring(title or ""); message = tostring(message or "")
    if activeNotifCount >= Config.NotifMaxVisible then
        if #queuedNotifs < 20 then
            table.insert(queuedNotifs, {title=title, message=message, duration=duration, kind=kind})
        end
        return
    end
    activeNotifCount = activeNotifCount + 1
    notifSerial = notifSerial + 1
    local meta = NotifMeta[kind] or NotifMeta.info
    local accent = meta.accent()
    local W = Config.NotifWidth

    local wrapper = Instance.new("Frame")
    wrapper.Name = "wrap_" .. notifSerial
    wrapper.Size = UDim2.new(1, 0, 0, 72)
    wrapper.BackgroundTransparency = 1

            if input.UserInputType == Enum.UserInputType.MouseButton1 then dismissNotif(entry) end
        end)
    end
    closeBtn.MouseButton1Click:Connect(function() dismissNotif(entry) end)
end

local function computeWmWidth()
    local total = 0
    if Config.WatermarkBackground then total = total + Config.WatermarkPadding * 2 end
    total = total + 12 + 8
    total = total + #Config.WatermarkText * (Config.WatermarkFontSize * 0.62) + 8
    if Config.WatermarkSub ~= "" then
        total = total + #Config.WatermarkSub * ((Config.WatermarkFontSize - 2) * 0.62)
    end
    if WmStatsOn() then total = total + 1 + 8 + 130 end
    return math.max(total, 120)
end

local function computeWmPos(size)
    local ox, oy = Config.WatermarkOffsetX, Config.WatermarkOffsetY
    local pos = Config.WatermarkPosition
    local w, h = size.X, size.Y
    if pos == "top-left" then return UDim2.new(0, ox, 0, oy)
    elseif pos == "top-center" then return UDim2.new(0.5, -w/2, 0, oy)
    elseif pos == "top-right" then return UDim2.new(1, -w - ox, 0, oy)
    elseif pos == "bottom-left" then return UDim2.new(0, ox, 1, -h - oy)
    elseif pos == "bottom-center" then return UDim2.new(0.5, -w/2, 1, -h - oy)
    elseif pos == "bottom-right" then return UDim2.new(1, -w - ox, 1, -h - oy)
    end
    return UDim2.new(0, ox, 0, oy)
end

                MainFrame.Visible = false

                safeTween(MiniPill, EASING.Spring, { Size = UDim2.new(0, 130, 0, 30) })

        MainFrame.Visible = true
        Shadow.Visible = true
        OuterGlow.Visible = true
        MainFrame.BackgroundTransparency = 0.4
        MainFrame.Size = UDim2.new(0, baseW * 0.92, 0, baseH * 0.92)

local iconRegistry = {}
local function drawIcon(holder, kind)
    holder.BackgroundTransparency = 1
    holder.Size = UDim2.new(0, 20, 0, 20)
    local parts = {}
    local function mkFrame(size, pos)
        local f = Instance.new("Frame")
        f.Size = size; f.Position = pos; f.BackgroundColor3 = Theme.TextDim
        f.BorderSizePixel = 0; f.ZIndex = 3; f.Parent = holder
        table.insert(parts, {obj=f, prop="BackgroundColor3"}); return f
    end
    local function mkStroke(obj, color)
        local s = stroke(obj, color or Theme.TextDim, 1.5)
        table.insert(parts, {obj=s, prop="Color"}); return s
    end
    if kind == "swords" then
        local a = mkFrame(UDim2.new(0,14,0,2), UDim2.new(0.5,-7,0.5,-1)); a.Rotation = 45; cornerAbs(a,1)
        local b = mkFrame(UDim2.new(0,14,0,2), UDim2.new(0.5,-7,0.5,-1)); b.Rotation = -45; cornerAbs(b,1)
    elseif kind == "eye" then
        local ring = mkFrame(UDim2.new(0,14,0,14), UDim2.new(0.5,-7,0.5,-7))
        ring.BackgroundTransparency = 1; cornerAbs(ring,7); mkStroke(ring)
        local dot = mkFrame(UDim2.new(0,6,0,6), UDim2.new(0.5,-3,0.5,-3)); cornerAbs(dot,3)
    elseif kind == "diamond" then
        local sq = mkFrame(UDim2.new(0,12,0,12), UDim2.new(0.5,-6,0.5,-6))
        sq.BackgroundTransparency = 1; sq.Rotation = 45; cornerAbs(sq,2); mkStroke(sq)
    elseif kind == "bars" then
        local b1 = mkFrame(UDim2.new(0,14,0,2), UDim2.new(0.5,-7,0.5,-6)); cornerAbs(b1,1)
        local b2 = mkFrame(UDim2.new(0,10,0,2), UDim2.new(0.5,-5,0.5,-1)); cornerAbs(b2,1)
        local b3 = mkFrame(UDim2.new(0,6,0,2), UDim2.new(0.5,-3,0.5,4)); cornerAbs(b3,1)
    elseif kind == "warn" then
        local tri = mkFrame(UDim2.new(0,14,0,14), UDim2.new(0.5,-7,0.5,-7))
        tri.BackgroundTransparency = 1; cornerAbs(tri,2); mkStroke(tri)
        local bang = mkFrame(UDim2.new(0,2,0,6), UDim2.new(0.5,-1,0.5,-5)); cornerAbs(bang,1)
        local dot = mkFrame(UDim2.new(0,2,0,2), UDim2.new(0.5,-1,0.5,3)); cornerAbs(dot,1)
    end
    iconRegistry[holder] = parts
    return parts
end

local function paintIcon(holder, active)
    local parts = iconRegistry[holder]
    if not parts then return end
    local target = active and Theme.Accent or Theme.TextDim
    for _, e in ipairs(parts) do
        if e.obj and e.obj.Parent then
            if e.prop == "Color" then pcall(function() e.obj.Color = target end)
            else pcall(function() e.obj[e.prop] = target end) end
        end
    end
end

        safeTween(btn, info, {

            BackgroundColor3 = active and Theme.Surface3 or Theme.Surface2,
        })

            Transparency = active and 0.3 or 1,

                if activeTabData ~= old then old.Canvas.Visible = false end
            end)
        end

        canvas.GroupTransparency = 1
        page.Position = UDim2.new(0, 0, 0, 10)
        safeTween(canvas, EASING.Slow, { GroupTransparency = 0 })
        safeTween(page, EASING.Slow, { Position = UDim2.new(0, 0, 0, 0) })
        PageTitle.Text = T(i18nKey)
        PageSub.Text = subtitleKey ~= "" and T(subtitleKey) or ""
    end)

    tabs[name] = tabData
    if not activeTabData then
        activeTabData = tabData
        canvas.Visible = true
        canvas.GroupTransparency = 0
        setActive(true, false)
        PageTitle.Text = T(i18nKey)
        PageSub.Text = subtitleKey ~= "" and T(subtitleKey) or ""
    end

    return canvas, page, tabData
end

        fill.Size = UDim2.new(pos, 0, 1, 0)

            safeTween(handle, EASING.Fast, { Size = UDim2.new(0, 12, 0, 12) })
        end
    end))
    trackConnection(UserInputService.InputChanged:Connect(function(input)
        if dragS and input.UserInputType == Enum.UserInputType.MouseMovement then update(input) end
    end))

        if not dragS then
            safeTween(f, EASING.Fast, { BackgroundColor3 = Theme.Hover })
            safeTween(fStroke, EASING.Fast, { Transparency = 0.1 })

            safeTween(fStroke, EASING.Fast, { Transparency = 0.3 })
        end

    local function closeThis()
        if currentPopup and currentPopup.Parent then
            local p = currentPopup
            safeTween(p, EASING.Fast, {
                BackgroundTransparency = 1,
                Size = UDim2.new(p.Size.X.Scale, p.Size.X.Offset, 0, 0),
            })
            task.delay(0.14, function() if p and p.Parent then p:Destroy() end end)
            for i = #openPopups, 1, -1 do
                if openPopups[i] == p then table.remove(openPopups, i) end
            end
        end
        currentPopup = nil
        chev.Text = "▾"
        if #openPopups == 0 then ClickCatcher.Visible = false end
    end

    local function open()
        closeAllPopups()
        local btnAbs = btn.AbsolutePosition
        local btnSize = btn.AbsoluteSize
        local listH = #options * 26 + 8
        local targetW = btnSize.X

        local pop = Instance.new("Frame")

            optBtn.BackgroundTransparency = 1

            optBtn.TextSize = 11
            optBtn.Font = Enum.Font.Gotham
            optBtn.TextXAlignment = Enum.TextXAlignment.Left
            optBtn.AutoButtonColor = false
            optBtn.ZIndex = 1001
            optBtn.Parent = pop
            corner(optBtn, 0.4)

            optBtn.MouseEnter:Connect(function()
                safeTween(optBtn, EASING.Fast, { BackgroundTransparency = 0.85, TextColor3 = Theme.Text })
            end)
            optBtn.MouseLeave:Connect(function()
                safeTween(optBtn, EASING.Fast, { BackgroundTransparency = 1, TextColor3 = opt == currentDefault and Theme.Accent or Theme.TextDim })
            end)
            optBtn.MouseButton1Click:Connect(function()
                btn.Text = "  " .. opt
                default = opt
                currentDefault = opt
                callback(opt)
                closeThis()
            end)
        end

        currentPopup = pop
        table.insert(openPopups, pop)
        ClickCatcher.Visible = true
        chev.Text = "▴"

        safeTween(pop, EASING.Med, { Size = UDim2.new(0, targetW, 0, listH), BackgroundTransparency = 0 })
        safeTween(popStroke, EASING.Med, { Transparency = 0 })
    end

local colorHistory = {}
local function pushColorHistory(c)
    for i = #colorHistory, 1, -1 do
        if colorHistory[i] == c then table.remove(colorHistory, i) end
    end
    table.insert(colorHistory, 1, c)
    while #colorHistory > 8 do table.remove(colorHistory) end
end

local function toHex(c)
    local r = math.floor(c.R * 255 + 0.5)
    local g = math.floor(c.G * 255 + 0.5)
    local b = math.floor(c.B * 255 + 0.5)
    return string.format("#%02X%02X%02X", r, g, b)
end

local function fromHex(s)
    s = s:gsub("#", "")
    if #s ~= 6 then return nil end
    local r = tonumber(s:sub(1,2), 16)
    local g = tonumber(s:sub(3,4), 16)
    local b = tonumber(s:sub(5,6), 16)
    if not (r and g and b) then return nil end
    return Color3.fromRGB(r, g, b)
end

    lbl.TextXAlignment = Enum.TextXAlignment.Left

    stroke(swatch, Theme.BorderHi, 1)

        safeTween(f, EASING.Fast, { BackgroundColor3 = Theme.Hover })
        safeTween(fStroke, EASING.Fast, { Transparency = 0.1 })

local function serializeConfig()
    local out = {}
    for k, v in pairs(Config) do
        if type(v) == "Color3" then
            out[k] = {__c3 = true, r = v.R, g = v.G, b = v.B}
        else
            out[k] = v
        end
    end
    return HttpService:JSONEncode(out)
end

local function deserializeIntoConfig(str)
    local ok, decoded = pcall(function() return HttpService:JSONDecode(str) end)
    if not ok or type(decoded) ~= "table" then return false end
    for k, v in pairs(decoded) do
        if type(v) == "table" and v.__c3 then
            Config[k] = Color3.new(v.r, v.g, v.b)
        else
            Config[k] = v
        end
    end
    return true
end

-- ===== CORE: HITBOX / DOGFIGHT / RADAR / ESP / SILENT AIM =====

local function getOwningPlayer(m)
    if not m then return nil end
    local p = Players:FindFirstChild(m.Name)
    if p and p:IsA("Player") then return p end
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr.Name == m.Name then return plr end
    end
    for _, plr in ipairs(Players:GetPlayers()) do
        if m.Name:find(plr.Name, 1, true) then return plr end
    end
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr.Character and (m == plr.Character or m:IsDescendantOf(plr.Character)) then return plr end
    end
    return nil
end

local function IsMyPlane(model)
    if not model then return false end
    local myModel = Workspace.Camera:FindFirstChild("plane_model")
    if myModel and (model == myModel or model:IsDescendantOf(myModel)) then return true end
    if model.Name == LP.Name then return true end
    if model.Name == tostring(LP.UserId) then return true end
    return false
end

local function ShouldShowModel(model, isBot)
    if not model then return false end
    if IsMyPlane(model) then return false end
    if isBot then return true end
    if not Config.TeamCheck then return true end
    local owner = getOwningPlayer(model)
    if not owner then return true end
    if owner == LP then return false end
    if owner.Team and LP.Team and owner.Team == LP.Team then return false end
    if owner.TeamColor and LP.TeamColor and owner.TeamColor == LP.TeamColor then return false end
    return true
end

-- HITBOX
local debugHighlights = {}
local function ClearDebugHighlights()
    for _, h in pairs(debugHighlights) do
        if h and h.Parent then h:Destroy() end
    end
    table.clear(debugHighlights)
end

local function ApplyRealHitbox()
    local isMouseDown = UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1)
    for _, part in ipairs(FindAllPlaneParts()) do
        local n = part.Name
        local model = part:FindFirstAncestorWhichIsA("Model")
        local show = model and ShouldShowModel(model)
        if (n == "Fuselage" or n == "LeftWing" or n == "RightWing") and show then
            local shouldExpand = true
            if Config.SmartHitbox and not isMouseDown then shouldExpand = false end
            local mul = (n == "Fuselage") and Config.HitboxFuselageMul or Config.HitboxWingMul
            if shouldExpand then
                if not State.OriginalSizes[part] then State.OriginalSizes[part] = part.Size end
                local orig = State.OriginalSizes[part]
                part.Size = Vector3.new(orig.X * mul, orig.Y, orig.Z * mul)
                if Config.HitboxDebug and not debugHighlights[part] then
                    local h = Instance.new("Highlight")
                    h.Adornee = part
                    h.FillColor = Color3.fromRGB(255, 220, 60)
                    h.OutlineColor = Color3.fromRGB(255, 140, 0)
                    h.FillTransparency = 0.7
                    h.OutlineTransparency = 0
                    h.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                    h.Parent = part
                    debugHighlights[part] = h
                end
            else
                if State.OriginalSizes[part] then part.Size = State.OriginalSizes[part] end
            end
        end
    end
    if not Config.HitboxDebug then ClearDebugHighlights() end
end

local function updateDogfightThreats(cam)
    table.clear(State.DogfightThreats)
    if not Config.MasterDogfight then return end
    local myChar = LP.Character
    if not myChar then return end
    local myHRP = myChar:FindFirstChild("HumanoidRootPart")
    if not myHRP then return end
    local myPos = myHRP.Position
    local _, models = FindAllPlaneParts()
    for _, data in ipairs(models) do
        local fus = data.fuselage
        local model = data.model
        if fus and fus.Parent and ShouldShowModel(model, data.isBot) then
            local enemyPos = fus.Position
            local enemyVel = fus.AssemblyLinearVelocity
            if enemyVel.Magnitude >= Config.DogfightMinSpeed then
                local toMe = myPos - enemyPos
                local dist = toMe.Magnitude
                if dist > 30 and dist < Config.DogfightRange then
                    local altDiff = math.abs(enemyPos.Y - myPos.Y)
                    if altDiff < Config.DogfightAltTolerance then
                        local toMeDir = toMe.Unit
                        local enemyDir = enemyVel.Unit
                        local dot = toMeDir:Dot(enemyDir)
                        if dot >= Config.DogfightDotThreshold then
                            local closing = enemyVel:Dot(toMeDir)
                            if closing > 10 then
                                local tti = dist / closing
                                if tti <= Config.DogfightTTI then
                                    table.insert(State.DogfightThreats, {
                                        pos = enemyPos, dist = dist, tti = tti,
                                        name = model.Name, isBot = data.isBot,
                                    })
                                end
                            end
                        end
                    end
                end
            end
        end
    end
end

local function DrawDogfightWarning(cam, vpSize, center)
    if not Config.MasterDogfight then return end
    local threats = State.DogfightThreats
    if #threats == 0 then return end
    local now = os.clock()
    local flash = 0.5 + 0.5 * math.sin(now * Config.DogfightFlashSpeed * math.pi)
    local col = Config.DogfightColor
    local closest = threats[1]
    for _, t in ipairs(threats) do
        if t.tti < closest.tti then closest = t end
    end
    if Config.DogfightShowBorder then
        local thickness = 6 + flash * 4
        local alpha = 0.35 + flash * 0.35
        local edges = {
            {Vector2.new(0, 2), Vector2.new(vpSize.X, 2)},
            {Vector2.new(0, vpSize.Y-2), Vector2.new(vpSize.X, vpSize.Y-2)},
            {Vector2.new(2, 0), Vector2.new(2, vpSize.Y)},
            {Vector2.new(vpSize.X-2, 0), Vector2.new(vpSize.X-2, vpSize.Y)},
        }
        for _, e in ipairs(edges) do
            local l = GetPooledDrawing("Line")
            l.Visible = true; l.From = e[1]; l.To = e[2]
            l.Color = col; l.Thickness = thickness; l.Transparency = 1 - alpha
        end
    end
    if Config.DogfightShowText then
        local textY = vpSize.Y * 0.15
        local textCol = col:Lerp(Color3.new(1,1,1), flash * 0.6)
        local glow = GetPooledDrawing("Text")
        glow.Visible = true; glow.Position = Vector2.new(center.X, textY); glow.Center = true
        glow.Outline = false; glow.Color = col; glow.Size = 30
        glow.Transparency = 0.6 + (1 - flash) * 0.3
        glow.Text = "DOGFIGHT"
        local txt = GetPooledDrawing("Text")
        txt.Visible = true; txt.Position = Vector2.new(center.X, textY); txt.Center = true
        txt.Outline = true; txt.Color = textCol; txt.Size = 24
        txt.Transparency = 0; txt.Text = "DOGFIGHT"
        local subTxt = GetPooledDrawing("Text")
        subTxt.Visible = true; subTxt.Position = Vector2.new(center.X, textY + 22)
        subTxt.Center = true; subTxt.Outline = true
        subTxt.Color = col:Lerp(Color3.new(1,1,1), 0.4); subTxt.Size = 12; subTxt.Transparency = 0
        subTxt.Text = string.format("%s — %.0fm — T-%.1fs", string.upper(tostring(closest.name)), closest.dist, math.max(closest.tti, 0))
    end
    if Config.DogfightShowArrow then
        for _, t in ipairs(threats) do
            local sp, on = cam:WorldToViewportPoint(t.pos)
            local tx, ty
            if on and sp.Z > 0 and sp.X > 40 and sp.X < vpSize.X - 40 and sp.Y > 40 and sp.Y < vpSize.Y - 40 then
                tx, ty = sp.X, sp.Y
            else
                local rel = t.pos - cam.CFrame.Position
                local camCF = cam.CFrame
                local rx = rel:Dot(camCF.RightVector)
                local ry = rel:Dot(camCF.UpVector)
                local ang = math.atan2(ry, rx)
                local edge = math.min(vpSize.X, vpSize.Y) * 0.40
                tx = center.X + math.cos(ang) * edge
                ty = center.Y - math.sin(ang) * edge
            end
            local dx = tx - center.X
            local dy = ty - center.Y
            local mag = math.sqrt(dx*dx + dy*dy)
            if mag > 1 then
                dx = dx / mag; dy = dy / mag
                local arrowLen = 22 + flash * 6
                local baseX = tx - dx * arrowLen
                local baseY = ty - dy * arrowLen
                local perpX = -dy; local perpY = dx
                local wingSize = 10
                local leftTip = Vector2.new(baseX + perpX * wingSize, baseY + perpY * wingSize)
                local rightTip = Vector2.new(baseX - perpX * wingSize, baseY - perpY * wingSize)
                local tip = Vector2.new(tx, ty)
                local l1 = GetPooledDrawing("Line")
                l1.Visible = true; l1.From = leftTip; l1.To = tip
                l1.Color = col; l1.Thickness = 3; l1.Transparency = 0
                local l2 = GetPooledDrawing("Line")
                l2.Visible = true; l2.From = rightTip; l2.To = tip
                l2.Color = col; l2.Thickness = 3; l2.Transparency = 0
                local t2 = GetPooledDrawing("Text")
                t2.Visible = true
                t2.Position = Vector2.new(baseX - dx * 16, baseY - dy * 16)
                t2.Center = true; t2.Outline = true
                t2.Color = col; t2.Size = 11; t2.Transparency = 0
                t2.Text = string.format("%.0fm", t.dist)
            end
        end
    end
end

-- RADAR
local function DrawRadar(cam, vpSize)
    if not Config.MasterRadar then return end
    local radarSize = Config.RadarSize
    local radius = radarSize / 2 - 6
    local center = Vector2.new(vpSize.X - radarSize / 2 - Config.RadarPosX, radarSize / 2 + Config.RadarPosY)
    local shape = Config.RadarShape or "Circle"
    if shape == "Square" then
        local bg = GetPooledDrawing("Square")
        bg.Visible = true; bg.Size = Vector2.new(radius * 2, radius * 2)
        bg.Position = Vector2.new(center.X - radius, center.Y - radius)
        bg.Color = Config.RadarBgColor; bg.Filled = true; bg.Thickness = 1; bg.Transparency = 0.2
        local ring = GetPooledDrawing("Square")
        ring.Visible = true; ring.Size = Vector2.new(radius * 2, radius * 2)
        ring.Position = Vector2.new(center.X - radius, center.Y - radius)
        ring.Color = Theme.BorderHi; ring.Filled = false; ring.Thickness = 1.5; ring.Transparency = 0.4
    else
        local bg = GetPooledDrawing("Circle")
        bg.Visible = true; bg.Position = center; bg.Radius = radius
        bg.Color = Config.RadarBgColor; bg.Filled = true; bg.Thickness = 1; bg.Transparency = 0.2
        for _, factor in ipairs({1.0, 0.66, 0.33}) do
            local r = GetPooledDrawing("Circle")
            r.Visible = true; r.Position = center; r.Radius = radius * factor
            r.Color = Theme.BorderHi; r.Filled = false; r.Thickness = 1; r.Transparency = 0.5
        end
    end
    local hLine = GetPooledDrawing("Line")
    hLine.Visible = true; hLine.From = Vector2.new(center.X - radius, center.Y); hLine.To = Vector2.new(center.X + radius, center.Y)
    hLine.Color = Theme.Border; hLine.Thickness = 1; hLine.Transparency = 0.65
    local vLine = GetPooledDrawing("Line")
    vLine.Visible = true; vLine.From = Vector2.new(center.X, center.Y - radius); vLine.To = Vector2.new(center.X, center.Y + radius)
    vLine.Color = Theme.Border; vLine.Thickness = 1; vLine.Transparency = 0.65
    local sweepAngle = (os.clock() * 1.6) % (math.pi * 2)
    local sweep = GetPooledDrawing("Line")
    sweep.Visible = true; sweep.From = center
    sweep.To = Vector2.new(center.X + math.cos(sweepAngle) * radius, center.Y + math.sin(sweepAngle) * radius)
    sweep.Color = Theme.Accent; sweep.Thickness = 2; sweep.Transparency = 0.2
    for k = 1, 10 do
        local a = sweepAngle - k * 0.09
        local trailLen = math.min(radius * (1 - k * 0.03), radius)
        local trl = GetPooledDrawing("Line")
        trl.Visible = true; trl.From = center
        trl.To = Vector2.new(center.X + math.cos(a) * trailLen, center.Y + math.sin(a) * trailLen)
        trl.Color = Theme.Accent; trl.Thickness = 1; trl.Transparency = 0.2 + k * 0.07
    end
    if Config.RadarShowCardinals and shape == "Circle" then
        local labels = {
            {t = "N", x = center.X, y = center.Y - radius - 10},
            {t = "S", x = center.X, y = center.Y + radius + 2},
            {t = "E", x = center.X + radius + 4, y = center.Y - 6},
            {t = "W", x = center.X - radius - 12, y = center.Y - 6},
        }
        for _, l in ipairs(labels) do
            local t = GetPooledDrawing("Text")
            t.Visible = true; t.Position = Vector2.new(l.x, l.y)
            t.Center = true; t.Outline = true
            t.Color = Theme.Accent; t.Size = 10; t.Text = l.t
        end
    end
    if Config.RadarShowRangeText then
        local rt = GetPooledDrawing("Text")
        rt.Visible = true; rt.Position = Vector2.new(center.X, center.Y + radius + 16)
        rt.Center = true; rt.Outline = true; rt.Color = Theme.TextDim; rt.Size = 10
        rt.Text = string.format("%dm", Config.RadarRange)
    end
    if Config.RadarShowBlips then
        local _, models = FindAllPlaneParts()
        local camCF = cam.CFrame
        local camPos = camCF.Position
        local range = Config.RadarRange
        local now = os.clock()
        for _, data in ipairs(models) do
            local model = data.model
            local fus = data.fuselage
            if fus and fus.Parent and ShouldShowModel(model, data.isBot) then
                local rel = fus.Position - camPos
                local flat = Vector3.new(rel.X, 0, rel.Z)
                local rX = flat:Dot(camCF.RightVector)
                local rZ = flat:Dot(camCF.LookVector)
                local dist = math.sqrt(rX*rX + rZ*rZ)
                if dist <= range then
                    local nx = rX / range
                    local ny = rZ / range
                    local bx = center.X + nx * radius
                    local by = center.Y - ny * radius
                    if Config.RadarShowAltitude then
                        local altDiff = fus.Position.Y - camPos.Y
                        by = by - math.clamp(altDiff / 200, -6, 6)
                    end
                    local blipSize = math.clamp(5 * (1 - dist / range) + 2, 2, 6)
                    local blip = GetPooledDrawing("Circle")
                    blip.Visible = true; blip.Position = Vector2.new(bx, by); blip.Radius = blipSize
                    blip.Color = Config.RadarBlipColor; blip.Filled = true; blip.Thickness = 1; blip.Transparency = 0
                    local ring = GetPooledDrawing("Circle")
                    ring.Visible = true; ring.Position = Vector2.new(bx, by); ring.Radius = blipSize + 2
                    ring.Color = Config.RadarBlipColor; ring.Filled = false; ring.Thickness = 1; ring.Transparency = 0.55
                    if Config.RadarShowPulse then
                        local pulsePhase = (now * 1.5) % 1
                        local pulse = GetPooledDrawing("Circle")
                        pulse.Visible = true; pulse.Position = Vector2.new(bx, by)
                        pulse.Radius = blipSize + 2 + pulsePhase * 8
                        pulse.Color = Config.RadarBlipColor; pulse.Filled = false; pulse.Thickness = 1
                        pulse.Transparency = 0.15 + pulsePhase * 0.7
                    end
                    if Config.RadarShowCones then
                        local look = fus.CFrame.LookVector
                        local flatLook = Vector3.new(look.X, 0, look.Z)
                        if flatLook.Magnitude > 0.01 then
                            flatLook = flatLook.Unit
                            local lx = flatLook:Dot(camCF.RightVector)
                            local lz = flatLook:Dot(camCF.LookVector)
                            local cone = GetPooledDrawing("Line")
                            cone.Visible = true; cone.From = Vector2.new(bx, by)
                            cone.To = Vector2.new(bx + lx * (blipSize + 8), by - lz * (blipSize + 8))
                            cone.Color = Config.RadarBlipColor; cone.Thickness = 1.5; cone.Transparency = 0.4
                        end
                    end
                end
            end
        end
    end
    if Config.RadarShowSelf then
        local sm = GetPooledDrawing("Circle")
        sm.Visible = true; sm.Position = center; sm.Radius = 3
        sm.Color = Theme.Good; sm.Filled = true; sm.Thickness = 1; sm.Transparency = 0
    end
    if Config.RadarShowHeading then
        local fwd = cam.CFrame.LookVector
        local fwdFlat = Vector3.new(fwd.X, 0, fwd.Z)
        if fwdFlat.Magnitude > 0.01 then
            fwdFlat = fwdFlat.Unit
            local head = GetPooledDrawing("Line")
            head.Visible = true; head.From = center
            head.To = Vector2.new(center.X + fwdFlat.X * 14, center.Y - fwdFlat.Z * 14)
            head.Color = Theme.Good; head.Thickness = 2; head.Transparency = 0
        end
    end
end

-- ESP
local function DrawESP(cam, vpSize, center)
    if not Config.MasterESP then return end
    local _, models = FindAllPlaneParts()
    local now = os.clock()
    local function inDistance(d)
        if d < Config.ESPMinDistance then return false end
        if d > Config.ESPMaxDistance then return false end
        return true
    end
    for _, data in ipairs(models) do
        local model = data.model
        local fuselage = data.fuselage
        if fuselage and fuselage.Parent and ShouldShowModel(model, data.isBot) then
            local anchorPos = fuselage.Position
            local distance = (anchorPos - cam.CFrame.Position).Magnitude
            if inDistance(distance) then
                local pos, onScreen = cam:WorldToViewportPoint(anchorPos)
                local inFov = true
                if Config.ESPFovOnly then
                    inFov = onScreen and pos.Z > 0 and ((Vector2.new(pos.X, pos.Y) - center).Magnitude <= Config.FOVRadius)
                end
                if onScreen and pos.Z > 0 and inFov then
                    local boxSize = math.clamp(2800 / math.max(distance, 50), 22, 90)
                    local bt = Config.ESPBoxThickness
                    local col = Config.ESPColor
                    if Config.ESPBox then
                        local box = GetPooledDrawing("Square")
                        box.Visible = true
                        box.Size = Vector2.new(boxSize, boxSize)
                        box.Position = Vector2.new(pos.X - boxSize/2, pos.Y - boxSize/2)
                        box.Color = col; box.Thickness = bt; box.Filled = false; box.Transparency = 0.15
                    end
                    if Config.ESPCorner then
                        local half = boxSize / 2
                        local seg = boxSize * 0.26
                        local corners = {
                            {pos.X-half, pos.Y-half, 1, 1},
                            {pos.X+half, pos.Y-half, -1, 1},
                            {pos.X-half, pos.Y+half, 1, -1},
                            {pos.X+half, pos.Y+half, -1, -1},
                        }
                        for _, c in ipairs(corners) do
                            local cx, cy, dx, dy = c[1], c[2], c[3], c[4]
                            local h = GetPooledDrawing("Line")
                            h.From = Vector2.new(cx, cy); h.To = Vector2.new(cx + seg*dx, cy)
                            h.Color = col; h.Thickness = bt+0.5; h.Transparency = 0
                            local v = GetPooledDrawing("Line")
                            v.From = Vector2.new(cx, cy); v.To = Vector2.new(cx, cy + seg*dy)
                            v.Color = col; v.Thickness = bt+0.5; v.Transparency = 0
                        end
                    end
                    if Config.ESPHealthBar then
                        local frac = 1.0
                        local hum = nil
                        local char = nil
                        for _, plr in ipairs(Players:GetPlayers()) do
                            if model.Name:find(plr.Name) then char = plr.Character; break end
                        end
                        if char then hum = char:FindFirstChildWhichIsA("Humanoid") end
                        if hum then frac = hum.Health / math.max(hum.MaxHealth, 1) end
                        local c2 = healthColor(frac)
                        if Config.ESPHealthPulse and frac < 0.3 then
                            local pulse = 0.7 + 0.3 * math.sin(now * 8)
                            c2 = c2:Lerp(Color3.new(1,1,1), 1 - pulse)
                        end
                        if Config.ESPHealthHorizontal then
                            local bw, bh = 48, 3
                            local bx = pos.X - bw/2
                            local by = pos.Y + boxSize/2 + 6
                            local bg = GetPooledDrawing("Square")
                            bg.Size = Vector2.new(bw, bh); bg.Position = Vector2.new(bx, by)
                            bg.Color = Color3.fromRGB(15,15,15); bg.Filled = true; bg.Transparency = 0.2
                            local fill = GetPooledDrawing("Square")
                            fill.Size = Vector2.new(bw * frac, bh); fill.Position = Vector2.new(bx, by)
                            fill.Color = c2; fill.Filled = true
                        else
                            local barH, barW = 44, 3
                            local barX = pos.X - boxSize/2 - 7
                            local barY = pos.Y - barH/2
                            local bg = GetPooledDrawing("Square")
                            bg.Size = Vector2.new(barW, barH); bg.Position = Vector2.new(barX, barY)
                            bg.Color = Color3.fromRGB(15,15,15); bg.Filled = true; bg.Transparency = 0.2
                            local fill = GetPooledDrawing("Square")
                            fill.Size = Vector2.new(barW, barH * frac)
                            fill.Position = Vector2.new(barX, barY + barH * (1 - frac))
                            fill.Color = c2; fill.Filled = true
                        end
                    end
                    if Config.ESPName or Config.ESPDistance then
                        local nameText = model.Name
                        if Config.ESPNameUpper then nameText = nameText:upper() end
                        if Config.ESPName then
                            local t = GetPooledDrawing("Text")
                            t.Position = Vector2.new(pos.X, pos.Y - boxSize/2 - 16)
                            t.Center = true; t.Outline = Config.ESPOutline
                            t.Color = col:Lerp(Color3.new(1,1,1), 0.5); t.Size = 11
                            t.Text = nameText
                        end
                        if Config.ESPDistance then
                            local t2 = GetPooledDrawing("Text")
                            t2.Position = Vector2.new(pos.X, pos.Y - boxSize/2 - 3)
                            t2.Center = true; t2.Outline = Config.ESPOutline
                            t2.Color = Color3.fromRGB(200,200,205); t2.Size = 10
                            t2.Text = string.format("[%dm]", math.floor(distance))
                        end
                    end
                elseif Config.ESPOffscreen then
                    local rel = anchorPos - cam.CFrame.Position
                    local camCF = cam.CFrame
                    local rx = rel:Dot(camCF.RightVector)
                    local ry = rel:Dot(camCF.UpVector)
                    local ang = math.atan2(ry, rx)
                    local edge = math.min(vpSize.X, vpSize.Y) * 0.40
                    local ex = center.X + math.cos(ang) * edge
                    local ey = center.Y - math.sin(ang) * edge
                    local dx = ex - center.X
                    local dy = ey - center.Y
                    local mag = math.sqrt(dx*dx + dy*dy)
                    if mag > 1 then
                        dx = dx / mag; dy = dy / mag
                        local big = Config.ESPOffscreenBig
                        local arrowLen = big and 26 or 16
                        local wingSize = big and 12 or 7
                        local bx = ex - dx * arrowLen
                        local by = ey - dy * arrowLen
                        local px = -dy; local py = dx
                        local l1 = GetPooledDrawing("Line")
                        l1.Visible = true
                        l1.From = Vector2.new(bx + px*wingSize, by + py*wingSize)
                        l1.To = Vector2.new(ex, ey)
                        l1.Color = col; l1.Thickness = big and 3.5 or 2.5; l1.Transparency = 0
                        local l2 = GetPooledDrawing("Line")
                        l2.Visible = true
                        l2.From = Vector2.new(bx - px*wingSize, by - py*wingSize)
                        l2.To = Vector2.new(ex, ey)
                        l2.Color = col; l2.Thickness = big and 3.5 or 2.5; l2.Transparency = 0
                        if Config.ESPOffscreenShowDistance then
                            local t = GetPooledDrawing("Text")
                            t.Visible = true
                            t.Position = Vector2.new(bx - dx*16, by - dy*16)
                            t.Center = true; t.Outline = true
                            t.Color = col; t.Size = big and 11 or 10
                            t.Text = string.format("%.0fm", distance)
                        end
                    end
                end
            end
        end
    end
end

-- PROTECTION
task.spawn(function()
    while State.Running do
        RunService.Heartbeat:Wait()
        if not Config.MasterProtection then continue end
        local char = LP.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if hrp then
            if Config.AntiVoid and hrp.Position.Y < -50 then
                hrp.CFrame = CFrame.new(hrp.Position.X, 500, hrp.Position.Z)
            end
            if Config.AntiFling then
                local vel = hrp.AssemblyLinearVelocity
                if vel.Magnitude > 500 then hrp.AssemblyLinearVelocity = vel.Unit * 100 end
            end
        end
    end
end)

-- SILENT AIM
task.spawn(function()
    if not game:IsLoaded() then game.Loaded:Wait() end
    local PS = LP:WaitForChild("PlayerScripts", 10)
    local CC = PS and PS:WaitForChild("ClientCore", 10)
    local FW = CC and CC:WaitForChild("Framework", 10)
    local MD = FW and FW:WaitForChild("Modules", 10)
    if not MD then return end

    local function safeRequire(o)
        if o and o:IsA("ModuleScript") then
            local ok, r = pcall(require, o)
            if ok then return r end
        end
        return nil
    end

    local Weapon = safeRequire(MD:FindFirstChild("Weapon"))
    local Controls = safeRequire(MD:FindFirstChild("Controls"))
    local CameraModule = safeRequire(MD:FindFirstChild("Camera"))
    local Plane = safeRequire(MD:FindFirstChild("Plane"))
    local ProjectileCompute = safeRequire(ReplicatedStorage.Shared:FindFirstChild("ProjectileParallelCompute"))
    local ArmamentData = safeRequire(ReplicatedStorage.Assets.Modules.Essentials:FindFirstChild("ArmamentData"))

    if not (Weapon and Controls and CameraModule and Plane and ProjectileCompute and ArmamentData) then
        warn("[WOG] silent aim: module require failed")
        return
    end
    State.SilentAimReady = true

    local ConditionService
    pcall(function() ConditionService = _G:require("ConditionService") end)

    State.OriginalNewBullet = Weapon.new_bullet
    State.WeaponModule = Weapon

    local function solveInterceptTime(shooterPos, targetPos, targetDir, targetSpeed, projectileSpeed, projectileMass)
        local speed = math.max(projectileSpeed, 0.001)
        local dragFactor = 0.02 / math.max(projectileMass, 1e-6)
        local delta = targetPos - shooterPos
        local t = delta.Magnitude / speed
        for _ = 1, 6 do
            local futureDistance = (delta + (targetDir * targetSpeed * t)).Magnitude
            local dragDistance = dragFactor * futureDistance
            local newTime
            if dragDistance > 1e-4 then
                newTime = (math.exp(math.min(dragDistance, 30)) - 1) / (dragFactor * speed)
            else
                newTime = futureDistance / speed
            end
            t = newTime
            if newTime >= 2.4 then return 2.4 end
        end
        return t
    end

    local function resolveTargetModel(name)
        if not name then return nil end
        local pf = CameraModule.plane_part_viewmodels
        if pf and pf:FindFirstChild(name) then return pf:FindFirstChild(name) end
        local bf = CameraModule.bot_plane_viewmodels
        if bf and bf:FindFirstChild(name) then return bf:FindFirstChild(name) end
        local vm = CameraModule.plane_viewmodels
        if vm and vm:FindFirstChild(name) then return vm:FindFirstChild(name) end
        return nil
    end

    local function getTargetPart(model)
        if not model then return nil end
        local chosen = model:FindFirstChild(Config.HitPart)
        if chosen and chosen:IsA("BasePart") then return chosen end
        if model.PrimaryPart then return model.PrimaryPart end
        return nil
    end

    local function enemyCheck(name)
        if not Config.TeamCheck then return true end
        local p = Players:FindFirstChild(name)
        if not p then return true end
        return p.TeamColor ~= LP.TeamColor
    end

    local function targetAlive(name)
        if ConditionService then
            local ok, destroyed = pcall(function() return ConditionService:getConditionValue(name, "destroyed") end)
            if ok and destroyed then return false end
        end
        return true
    end

    local function insideFOV(worldPos)
        local cam = Workspace.CurrentCamera
        if not cam then return false, math.huge end
        local p, visible = cam:WorldToViewportPoint(worldPos)
        if not visible or p.Z <= 0 then return false, math.huge end
        local center = cam.ViewportSize * 0.5
        local dist = (Vector2.new(p.X, p.Y) - center).Magnitude
        return dist <= Config.FOVRadius, dist
    end

    local function updateTargetMotion(name, position)
        local now = os.clock()
        local h = State.History[name]
        if not h then
            State.History[name] = {pos = position, time = now, smoothVelocity = Vector3.zero}
            return Vector3.zero, 0
        end
        local dt = math.clamp(now - h.time, 1/240, 0.25)
        local rawVelocity = (position - h.pos) / dt
        local alpha = 1 - math.exp(-dt / 0.15)
        h.smoothVelocity = h.smoothVelocity:Lerp(rawVelocity, alpha)
        h.pos = position
        h.time = now
        local direction = h.smoothVelocity.Magnitude > 1e-4 and h.smoothVelocity.Unit or Vector3.zero
        local targetSpeed = h.smoothVelocity.Magnitude
        if ConditionService then
            local ok, speed = pcall(function() return ConditionService:getConditionValue(name, "speed") end)
            if ok and tonumber(speed) then targetSpeed = tonumber(speed) / 14.4 end
        end
        return direction, targetSpeed
    end

    local function acquireGameTarget(force)
        local now = os.clock()
        if not force and now - State.LastAcquire < Config.TargetSwitchCooldown then return end
        State.LastAcquire = now
        local folders = {CameraModule.plane_part_viewmodels, CameraModule.bot_plane_viewmodels, CameraModule.plane_viewmodels}

        if Config.TargetSelection == "Closest Distance" then
            local closestName, minDist = nil, math.huge
            local origin = Workspace.CurrentCamera and Workspace.CurrentCamera.CFrame.Position or Vector3.zero
            for _, folder in ipairs(folders) do
                if folder then
                    for _, model in ipairs(folder:GetChildren()) do
                        if model:IsA("Model") and model.Name ~= LP.Name and enemyCheck(model.Name) and targetAlive(model.Name) then
                            local part = getTargetPart(model)
                            if part then
                                local d = (part.Position - origin).Magnitude
                                if d < minDist then minDist = d; closestName = model.Name end
                            end
                        end
                    end
                end
            end
            if closestName then
                State.CurrentTarget = closestName
                State.CurrentTargetModel = resolveTargetModel(closestName)
                State.LastTargetChange = now
                return
            end
        elseif Config.TargetSelection == "Closest FOV" then
            local closestName, minFovDist = nil, math.huge
            for _, folder in ipairs(folders) do
                if folder then
                    for _, model in ipairs(folder:GetChildren()) do
                        if model:IsA("Model") and model.Name ~= LP.Name and enemyCheck(model.Name) and targetAlive(model.Name) then
                            local part = getTargetPart(model)
                            if part then
                                local inFov, fovDist = insideFOV(part.Position)
                                if inFov and fovDist < minFovDist then minFovDist = fovDist; closestName = model.Name end
                            end
                        end
                    end
                end
            end
            if closestName then
                State.CurrentTarget = closestName
                State.CurrentTargetModel = resolveTargetModel(closestName)
                State.LastTargetChange = now
                return
            end
        end

        pcall(function() Controls:lock_on_plane("auto") end)
        local name = Controls.AimAssistTarget or Controls.PlayerLocked
        if name and name ~= false and name ~= "" then
            name = tostring(name)
            if State.CurrentTarget ~= name then
                if State.CurrentTarget and now - State.LastTargetChange < Config.TargetHoldTime and not force then return end
                State.CurrentTarget = name
                State.CurrentTargetModel = resolveTargetModel(name)
                State.LastTargetChange = now
            end
        end
    end

    local function validateTarget(origin)
        if not State.CurrentTarget then return nil end
        if not enemyCheck(State.CurrentTarget) then return nil end
        if not targetAlive(State.CurrentTarget) then return nil end
        local model = State.CurrentTargetModel or resolveTargetModel(State.CurrentTarget)
        State.CurrentTargetModel = model
        if not model then return nil end
        local part = getTargetPart(model)
        if not part then return nil end
        local distance = (part.Position - origin).Magnitude
        State.TargetDistance = distance
        if distance > Config.MaxDistance then return nil end
        local inFov, screenDistance = insideFOV(part.Position)
        if not inFov then return nil end
        return part
    end

    local function simulateProjectile(startCF, velocity, mass, flightTime)
        if typeof(startCF) ~= "CFrame" then return nil end
        velocity = tonumber(velocity); mass = tonumber(mass); flightTime = tonumber(flightTime)
        if not velocity or not mass or not flightTime or flightTime <= 0 then return nil end
        local cf, speed = startCF, velocity
        local startPos = startCF.Position
        local elapsed = 0
        while elapsed < flightTime do
            local dt = math.min(1/120, flightTime - elapsed)
            local ok, newCF, _, newVelocity = pcall(ProjectileCompute.integrate, cf, 0, -0.12, speed, mass, startPos, dt)
            if not ok or typeof(newCF) ~= "CFrame" or type(newVelocity) ~= "number" then return nil end
            cf, speed = newCF, newVelocity
            elapsed = elapsed + dt
        end
        return cf.Position
    end

    local function buildSilentAimCFrame(originalCFrame, bulletData, gunType)
        if not Config.MasterSilentAim then return originalCFrame, nil end
        if typeof(originalCFrame) ~= "CFrame" then return originalCFrame, nil end
        acquireGameTarget(false)
        local targetPart = validateTarget(originalCFrame.Position)
        local gunData = gunType and ArmamentData[gunType]
        if targetPart and gunData and tonumber(gunData.DistLimit) then
            if State.TargetDistance > tonumber(gunData.DistLimit) then return originalCFrame, nil end
        end
        if not targetPart then
            State.CurrentTarget = nil
            State.CurrentTargetModel = nil
            acquireGameTarget(true)
            targetPart = validateTarget(originalCFrame.Position)
            if not targetPart then return originalCFrame, nil end
        end
        local targetPos = targetPart.Position
        local predictedPos = targetPos
        local targetDirection, targetSpeed = Vector3.zero, 0
        if Config.PredictionEnabled then
            targetDirection, targetSpeed = updateTargetMotion(State.CurrentTarget, targetPos)
            local projectile = bulletData and bulletData.projectile
            if projectile and tonumber(projectile.velocity) and tonumber(projectile.mass) then
                local projectileSpeed = ((projectile.velocity * 3.6) + (tonumber(Plane.total_speed) or 0)) / 14.4
                local interceptTime = solveInterceptTime(originalCFrame.Position, targetPos, targetDirection, targetSpeed, projectileSpeed, projectile.mass)
                predictedPos = targetPos + (targetDirection * targetSpeed * interceptTime)
                State.LastInterceptTime = interceptTime
            end
        end
        local projectile = bulletData and bulletData.projectile
        if not projectile then return originalCFrame, nil end
        local baseVelocity = tonumber(projectile.velocity)
        local projectileMass = tonumber(projectile.mass)
        if not baseVelocity or not projectileMass then return originalCFrame, nil end
        local projectileSpeed = ((baseVelocity * 3.6) + (tonumber(Plane.total_speed) or 0)) / 14.4
        local flightTime = State.LastInterceptTime > 0 and State.LastInterceptTime or (predictedPos - originalCFrame.Position).Magnitude / math.max(projectileSpeed, 0.001)
        local aimPoint = predictedPos
        if Config.BallisticCompensation then
            for _ = 1, 4 do
                local delta = aimPoint - originalCFrame.Position
                if delta.Magnitude <= 1e-4 then break end
                local testCF = CFrame.lookAt(originalCFrame.Position, aimPoint, originalCFrame.UpVector)
                local simPos = simulateProjectile(testCF, projectileSpeed, projectileMass, flightTime)
                if not simPos then break end
                local err = predictedPos - simPos
                aimPoint = aimPoint + err
                if err.Magnitude < 0.5 then break end
            end
        end
        local finalDir = aimPoint - originalCFrame.Position
        if finalDir.Magnitude <= 1e-4 then return originalCFrame, nil end
        return CFrame.lookAt(originalCFrame.Position, aimPoint, originalCFrame.UpVector), aimPoint
    end

    Weapon.new_bullet = function(self, a2, a3, a4, a5, a6, a7, a8, a9)
        local shotCFrame = a6
        local isLocal = (a2 == LP.Name)
        if State.Running and typeof(a6) == "CFrame" then
            local gunName = tostring(a7 or ""):lower()
            local isMissile = gunName:find("missile") or gunName:find("rocket") or
                              gunName:find("bomb") or gunName:find("torpedo") or
                              gunName:find("homing") or gunName:find("guided")
            if Config.MasterTracer and (Config.TracerTarget == "Everyone" or isLocal) and not isMissile then
                local endPos = a6.Position + (a6.LookVector * 3000)
                local hasHit = false
                if Config.MasterSilentAim and isLocal then
                    local ok, redirected, aimPoint = pcall(buildSilentAimCFrame, a6, a5, a7)
                    if ok and typeof(redirected) == "CFrame" then
                        shotCFrame = redirected
                        if aimPoint then endPos = aimPoint; hasHit = true end
                    end
                end
                if #State.ActiveTracers >= (Config.MaxTracers or 25) then
                    table.remove(State.ActiveTracers, 1)
                end
                table.insert(State.ActiveTracers, {
                    from = shotCFrame.Position, to = endPos,
                    created = os.clock(), isLocal = isLocal, hit = hasHit,
                })
            elseif Config.MasterSilentAim and isLocal then
                local ok, redirected = pcall(buildSilentAimCFrame, a6, a5, a7)
                if ok and typeof(redirected) == "CFrame" then shotCFrame = redirected end
            end
        end
        return State.OriginalNewBullet(self, a2, a3, a4, a5, shotCFrame, a7, a8, a9)
    end

    acquireGameTarget(true)
    print("[WOG] silent aim ready")
end)

-- PERF
function DisableShadows()
    pcall(function() State.SavedGlobalShadows = Lighting.GlobalShadows; Lighting.GlobalShadows = false end)
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("BasePart") then
            State.SavedParts[obj] = State.SavedParts[obj] or {}
            if State.SavedParts[obj].CastShadow == nil then State.SavedParts[obj].CastShadow = obj.CastShadow end
            obj.CastShadow = false
        end
    end
end
function RestoreShadows()
    if State.SavedGlobalShadows ~= nil then Lighting.GlobalShadows = State.SavedGlobalShadows end
    for p, d in pairs(State.SavedParts) do
        if p and p.Parent and d.CastShadow ~= nil then p.CastShadow = d.CastShadow end
    end
end
function DisableParticles()
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Smoke")
            or obj:IsA("Fire") or obj:IsA("Sparkles") or obj:IsA("Beam") then
            State.ThrottledParticles[obj] = obj.Enabled
            obj.Enabled = false
        end
    end
end
function RestoreParticles()
    for o, v in pairs(State.ThrottledParticles) do
        if o and o.Parent then o.Enabled = v end
    end
    table.clear(State.ThrottledParticles)
end
function ThrottleParticles()
    local f = Config.PerfParticleThrottle
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("ParticleEmitter") then
            if State.SavedParts[obj] == nil then State.SavedParts[obj] = obj.Rate end
            if type(State.SavedParts[obj]) == "number" then
                obj.Rate = math.max(0, State.SavedParts[obj] * f)
            end
        end
    end
end
function RestoreThrottledParticles() end
function DisableDecals()
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("Decal") or obj:IsA("Texture") then
            State.HiddenParts[obj] = obj.Transparency
            obj.Transparency = 1
        end
    end
end
function RestoreDecals()
    for o, v in pairs(State.HiddenParts) do
        if o and o.Parent then o.Transparency = v end
    end
    table.clear(State.HiddenParts)
end
function DisablePostFX()
    pcall(function()
        State.SavedQuality = settings().Rendering.QualityLevel
        settings().Rendering.QualityLevel = Enum.QualityLevel.Level1
    end)
    pcall(function() State.SavedBrightness = Lighting.Brightness; Lighting.Brightness = 2 end)
    for _, c in ipairs(Lighting:GetChildren()) do
        if c:IsA("PostEffect") or c:IsA("Atmosphere") then
            c.Enabled = false
        end
    end
end
function RestorePostFX()
    if State.SavedQuality then pcall(function() settings().Rendering.QualityLevel = State.SavedQuality end) end
    if State.SavedBrightness then Lighting.Brightness = State.SavedBrightness end
    for _, c in ipairs(Lighting:GetChildren()) do
        if c:IsA("PostEffect") or c:IsA("Atmosphere") then c.Enabled = true end
    end
end
function FlattenMaterials()
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("BasePart") then
            State.SavedParts[obj] = State.SavedParts[obj] or {}
            if State.SavedParts[obj].Material == nil then
                State.SavedParts[obj].Material = obj.Material
                State.SavedParts[obj].Reflectance = obj.Reflectance
            end
            pcall(function() obj.Material = Enum.Material.SmoothPlastic; obj.Reflectance = 0 end)
        end
    end
end
function RestoreMaterials()
    for p, d in pairs(State.SavedParts) do
        if p and p.Parent and type(d) == "table" then
            if d.Material ~= nil then pcall(function() p.Material = d.Material end) end
            if d.Reflectance ~= nil then pcall(function() p.Reflectance = d.Reflectance end) end
        end
    end
end
function HideFarParts()
    local cam = Workspace.CurrentCamera
    if not cam then return end
    local camPos = cam.CFrame.Position
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("BasePart") then
            local isPlane = obj.Name == "Fuselage" or obj.Name == "LeftWing" or obj.Name == "RightWing"
            if not isPlane then
                local d = (obj.Position - camPos).Magnitude
                if d > Config.PerfHideFarDistance and obj.Transparency < 1 then
                    State.HiddenParts[obj] = obj.Transparency
                    obj.Transparency = 1
                end
            end
        end
    end
end
function RestoreFarParts()
    for o, v in pairs(State.HiddenParts) do
        if o and o.Parent then o.Transparency = v end
    end
    table.clear(State.HiddenParts)
end

    updateDogfightThreats(cam)

    -- tracers
    if Config.MasterTracer then
        local maxLife = Config.TracerLifetime or 0.75
        local bt = Config.TracerThickness or 1.4
        local style = Config.TracerStyle or "Neon"
        local colA = Config.TracerColorA or Color3.fromRGB(0,255,230)
        local colB = Config.TracerColorB or Color3.fromRGB(180,60,255)
        local jitter = Config.TracerJitter or 6
        local rbSpeed = Config.TracerRainbowSpeed or 1.5
        local invMax = 1 / math.max(maxLife, 0.01)
        local camPos = cam.CFrame.Position
        local camLook = cam.CFrame.LookVector

        local function seg(v1, v2, c, th, tp)
            local l = GetPooledDrawing("Line")
            l.Visible = true; l.From = v1; l.To = v2
            l.Color = c; l.Thickness = th; l.Transparency = tp
        end

        for i = #State.ActiveTracers, 1, -1 do
            local tr = State.ActiveTracers[i]
            local age = now - tr.created
            if age > maxLife then
                table.remove(State.ActiveTracers, i)
            else
                local alpha = math.clamp(1 - age * invMax, 0, 1)
                local prog = math.clamp(age * invMax, 0, 1)
                local trCol = tr.isLocal and colA or Color3.fromRGB(255,120,40)
                local pS, pE = tr.from, tr.to
                local bTh = bt
                if style == "Projectile" or style == "Comet" then
                    pE = tr.from:Lerp(tr.to, math.min(prog * 1.5, 1))
                    pS = tr.from:Lerp(tr.to, math.max((prog * 1.5) - 0.3, 0))
                elseif style == "Pulse" then
                    bTh = bt * (0.6 + 0.4 * math.sin(age * 18))
                end
                local c1, c2 = clipSegmentToCamera(pS, pE, cam)
                if c1 and c2 then
                    local s1 = cam:WorldToViewportPoint(c1)
                    local s2 = cam:WorldToViewportPoint(c2)
                    if s1.Z > 0 and s2.Z > 0 then
                        local v1 = Vector2.new(s1.X, s1.Y)
                        local v2 = Vector2.new(s2.X, s2.Y)
                        if Config.TracerAfterglow then
                            seg(v1, v2, trCol, bTh * 6, alpha * 0.06)
                            seg(v1, v2, trCol, bTh * 3, alpha * 0.10)
                        end
                        if style == "Beam" then
                            seg(v1, v2, trCol, bTh, alpha)
                            seg(v1, v2, Color3.new(1,1,1), math.max(bTh*0.35,1), alpha*0.85)
                        elseif style == "Neon" then
                            seg(v1, v2, trCol, bTh*2.2, alpha*0.35)
                            seg(v1, v2, trCol, bTh, alpha*0.95)
                            seg(v1, v2, Color3.new(1,1,1), math.max(bTh*0.45,1), alpha)
                        elseif style == "Glow Line" then
                            seg(v1, v2, trCol, bTh*2.6, alpha*0.28)
                            seg(v1, v2, trCol, bTh, alpha*0.75)
                        elseif style == "Lightning" then
                            local prevX, prevY = v1.X, v1.Y
                            for s = 1, 5 do
                                local t = s / 5
                                local px = v1.X + (v2.X - v1.X) * t
                                local py = v1.Y + (v2.Y - v1.Y) * t
                                if s < 5 then
                                    px = px + (math.random() - 0.5) * jitter * (1 - prog * 0.4)
                                    py = py + (math.random() - 0.5) * jitter * (1 - prog * 0.4)
                                end
                                seg(Vector2.new(prevX, prevY), Vector2.new(px, py), trCol, bTh, alpha)
                                prevX, prevY = px, py
                            end
                        elseif style == "Pulse" then
                            seg(v1, v2, trCol, bTh*2, alpha*0.35)
                            seg(v1, v2, Color3.new(1,1,1), bTh, alpha)
                        elseif style == "Comet" then
                            for s = 0, 4 do

                                seg(

                                    Vector2.new(v1.X + (v2.X - v1.X) * t1, v1.Y + (v2.Y - v1.Y) * t1),
                                    trCol, bTh*(0.4+t1*1.1), 1-alpha*(t1*t1)
                                )
                            end

                                seg(
                                    Vector2.new(v1.X + (v2.X - v1.X) * t0, v1.Y + (v2.Y - v1.Y) * t0),

                            end
                        elseif style == "Rainbow" then
                            for s = 0, 5 do
                                local t0, t1 = s/6, (s+1)/6
                                seg(
                                    Vector2.new(v1.X + (v2.X - v1.X) * t0, v1.Y + (v2.Y - v1.Y) * t0),
                                    Vector2.new(v1.X + (v2.X - v1.X) * t1, v1.Y + (v2.Y - v1.Y) * t1),

                            end

                            local head = GetPooledDrawing("Circle")
